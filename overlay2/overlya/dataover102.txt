cat > /tmp/nmtest.sh <<'EOF'
#!/bin/bash
LOG=/home/alice/nmtest.log
CON="$(nmcli -g GENERAL.CONNECTION device show eth0)"
Y="$(grep -l gateway4 /etc/netplan/90-NM-*.yaml | head -1)"
OLD="$(ip -4 -o addr show eth0 | awk '{print $4}')"
snap() { echo "--- $1  $(date +%T)"; stat -c "mtime: %y" "$Y"; grep -A1 "^      addresses:" "$Y" | head -2; ip -4 -o addr show eth0 | awk '{print "ip:", $4}'; }
{
echo "connection=$CON  yaml=$Y  old=$OLD"
systemctl stop dip ipservice
snap BEFORE
nmcli connection modify "$CON" ipv4.addresses 192.168.137.150/24
nmcli connection down "$CON"; nmcli connection up "$CON"; sleep 8
snap AFTER_CHANGE
nmcli connection modify "$CON" ipv4.addresses "$OLD"
nmcli connection down "$CON"; nmcli connection up "$CON"; sleep 8
snap AFTER_REVERT
systemctl start dip ipservice
echo DONE
} > "$LOG" 2>&1
EOF
